# Westminster's yearly pulses for riots and the Crown (appended to vanilla's yearly_global_pulse).
yearly_global_pulse = {
	on_actions = {
		wm_riot_yearly_pulse
		wm_crown_yearly_pulse
	}
}

# No root. Tidies up riots whose organiser has gone, then (usually) starts a new riot somewhere,
# most likely in a seat whose people are unhappy with their MP.
wm_riot_yearly_pulse = {
	effect = {
		if = {
			limit = { has_global_variable_list = wm_active_riots }
			every_in_global_list = {
				variable = wm_active_riots
				limit = {
					NOT = { var:wm_riot_organiser ?= { is_alive = yes } }
				}
				add_to_temporary_list = wm_stale_riots
			}
			every_in_list = {
				list = wm_stale_riots
				wm_riot_end_effect = yes
			}
		}
		if = {
			limit = {
				exists = title:e_united_kingdom.holder
				NOT = {
					any_in_global_list = {
						variable = wm_active_riots
						count >= 2
					}
				}
			}
			random = {
				chance = wm_trust_riot_chance	# 75% at public trust 40; more when trust is low (wm_trust_values.txt)
				title:e_united_kingdom = {
					random_in_de_jure_hierarchy = {
						limit = {
							tier = tier_county
							exists = holder
							NOT = { wm_riot_is_active_trigger = yes }
						}
						weight = {
							base = 1
							modifier = {
								add = 4
								county_opinion < 0
							}
							modifier = {
								add = 10
								county_opinion < -30
							}
						}
						wm_riot_start_random_effect = yes
					}
				}
			}
		}
	}
}

# No root. Very rarely, an AI monarch with prestige to burn speaks out against an MP they dislike.
wm_crown_yearly_pulse = {
	effect = {
		title:k_wm_crown.holder ?= {
			if = {
				limit = {
					is_ai = yes
					prestige >= wm_crown_speak_out_prestige_cost
					NOT = { has_character_flag = wm_crown_spoke_out_recently }
				}
				save_scope_as = wm_monarch
				random = {
					chance = 3
					title:e_united_kingdom = {
						random_in_de_jure_hierarchy = {
							limit = {
								tier = tier_county
								exists = holder
								holder = {
									NOT = { this = scope:wm_monarch }
									reverse_opinion = {
										target = scope:wm_monarch
										value <= -20
									}
								}
							}
							holder = { save_scope_as = wm_speak_out_target }
						}
					}
					if = {
						limit = { exists = scope:wm_speak_out_target }
						scope:wm_monarch = { wm_crown_speak_out_by_script_effect = { TARGET = scope:wm_speak_out_target } }
					}
				}
			}
		}
	}
}

# History files can set variables and flags, but the game's "used but never set" check only counts script.
# Setting them again at game start keeps the error log clean (and costs nothing).
on_game_start = {
	on_actions = { wm_on_game_start }
}
wm_on_game_start = {
	effect = {
		every_living_character = {
			limit = { has_variable = wm_party }
			set_variable = { name = wm_party value = var:wm_party }
		}
		every_living_character = {
			limit = { has_character_flag = wm_speaker }
			add_character_flag = wm_speaker
		}
	}
}
