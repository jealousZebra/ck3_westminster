# The first Thatcher ministry and its opening political choice.
on_game_start_after_lobby = {
	on_actions = { wm_1979_bookmark_start }
}

wm_1979_bookmark_start = {
	trigger = {
		game_start_date >= 1979.5.4
		game_start_date < 1980.1.1
		exists = title:e_united_kingdom.holder
	}
	effect = {
		title:e_united_kingdom.holder = {
			# Vanilla fills the council before this hook. Clear those five posts
			# before installing the historical cabinet; keep the spouse's post.
			cp:councillor_steward ?= { save_temporary_scope_as = wm_outgoing_minister prev = { always_fire_councillor = scope:wm_outgoing_minister } }
			cp:councillor_spymaster ?= { save_temporary_scope_as = wm_outgoing_minister prev = { always_fire_councillor = scope:wm_outgoing_minister } }
			cp:councillor_marshal ?= { save_temporary_scope_as = wm_outgoing_minister prev = { always_fire_councillor = scope:wm_outgoing_minister } }
			cp:councillor_chancellor ?= { save_temporary_scope_as = wm_outgoing_minister prev = { always_fire_councillor = scope:wm_outgoing_minister } }
			cp:councillor_court_chaplain ?= { save_temporary_scope_as = wm_outgoing_minister prev = { always_fire_councillor = scope:wm_outgoing_minister } }
			assign_councillor_type = { type = councillor_steward target = character:wm_1979_geoffrey_howe }
			assign_councillor_type = { type = councillor_spymaster target = character:wm_1979_william_whitelaw }
			assign_councillor_type = { type = councillor_marshal target = character:wm_1979_john_nott }
			assign_councillor_type = { type = councillor_chancellor target = character:wm_1979_ian_gilmour }
			assign_councillor_type = { type = councillor_court_chaplain target = character:wm_1979_keith_joseph }
			trigger_event = wm_1979.1
		}
	}
}
