# Parties are houses. A character's party is var:wm_party (a k_party_* title) and their house is the party's house
# (wm_party_sync_house_effect, generated in wm_party_houses.txt by tools/data/build_2024.py). The decisions are in
# common/decisions/wm_party_decisions.txt.

# Character: crosses the floor to $PARTY$ (a party title with a leader). They take its whip and join its house; an
# MP serves the new party leader (or the Prime Minister if they lead it); an unlanded member joins the new leader's
# court. Their old party leader takes it badly.
wm_party_join_effect = {
	save_scope_as = wm_join_member
	clear_saved_scope = wm_join_liege
	$PARTY$ = { save_scope_as = wm_join_party }
	if = {
		limit = {
			exists = var:wm_party.holder
			NOT = { var:wm_party.holder = scope:wm_join_member }
		}
		var:wm_party.holder = {
			add_opinion = {
				target = scope:wm_join_member
				modifier = wm_party_crossed_floor_opinion
			}
		}
	}
	set_variable = { name = wm_party value = scope:wm_join_party }
	wm_party_fresh_start_effect = yes
	wm_party_sync_house_effect = yes
	if = {
		limit = {
			is_landed = yes
			NOT = { title:e_united_kingdom.holder ?= scope:wm_join_member }
		}
		if = {
			limit = {
				exists = scope:wm_join_party.holder
				NOT = { scope:wm_join_party.holder = scope:wm_join_member }
			}
			scope:wm_join_party.holder = { save_scope_as = wm_join_liege }
		}
		else = {
			title:e_united_kingdom.holder ?= { save_scope_as = wm_join_liege }
		}
		if = {
			limit = {
				exists = scope:wm_join_liege
				NOT = { liege ?= scope:wm_join_liege }
			}
			create_title_and_vassal_change = {
				type = swear_fealty
				save_scope_as = wm_join_change
				add_claim_on_loss = no
			}
			change_liege = {
				liege = scope:wm_join_liege
				change = scope:wm_join_change
			}
			resolve_title_and_vassal_change = scope:wm_join_change
		}
	}
	else_if = {
		limit = {
			is_ruler = no
			exists = scope:wm_join_party.holder
			NOT = { scope:wm_join_party.holder = scope:wm_join_member }
		}
		scope:wm_join_party.holder = { add_courtier = scope:wm_join_member }
	}
	wm_majority_update_effect = yes	# the government's working majority (wm_majority_effects.txt)
}

# Character: resigns the whip and sits as an independent (the Independent house); an MP serves the Prime Minister.
wm_party_leave_effect = {
	save_scope_as = wm_join_member
	if = {
		limit = {
			exists = var:wm_party.holder
			NOT = { var:wm_party.holder = scope:wm_join_member }
		}
		var:wm_party.holder = {
			add_opinion = {
				target = scope:wm_join_member
				modifier = wm_party_crossed_floor_opinion
			}
		}
	}
	remove_variable = wm_party
	wm_party_fresh_start_effect = yes
	wm_party_sync_house_effect = yes
	if = {
		limit = {
			is_landed = yes
			exists = title:e_united_kingdom.holder
			NOT = { title:e_united_kingdom.holder = scope:wm_join_member }
			NOT = { liege ?= title:e_united_kingdom.holder }
		}
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = wm_join_change
			add_claim_on_loss = no
		}
		change_liege = {
			liege = title:e_united_kingdom.holder
			change = scope:wm_join_change
		}
		resolve_title_and_vassal_change = scope:wm_join_change
	}
	wm_majority_update_effect = yes	# the government's working majority (wm_majority_effects.txt)
}

# Character: a new party, a clean slate. No rebellions against the old whip; a hopeful starts again at the bottom
# of the new party's approved list (unlanded careers, wm_career_effects.txt).
wm_party_fresh_start_effect = {
	remove_variable = wm_commons_rebellions
	remove_variable = wm_on_shortlist
	if = {
		limit = { has_variable = wm_standing }
		set_variable = { name = wm_standing value = 10 }
	}
}

# Character (an MP, not a party leader, the house head or the dynast): founds the party in slot $SLOT$
# (title:k_party_$SLOT$, uncreated or vacant). They serve the Prime Minister first (a party leader is a kingdom-tier
# ruler under the PM), then get a new dynasty and house named after the party with its made-up emblem
# (wm_coa_party_$SLOT$, tools/ui/build_party_arms.py), and the party title with an elected leadership.
wm_party_found_in_slot_effect = {
	save_scope_as = wm_found_founder
	title:k_party_$SLOT$ = { save_scope_as = wm_found_party }
	if = {
		limit = {
			exists = var:wm_party.holder
			NOT = { var:wm_party.holder = scope:wm_found_founder }
		}
		var:wm_party.holder = {
			add_opinion = {
				target = scope:wm_found_founder
				modifier = wm_party_split_opinion
			}
		}
	}
	if = {
		limit = {
			exists = title:e_united_kingdom.holder
			NOT = { title:e_united_kingdom.holder = scope:wm_found_founder }
			NOT = { liege ?= title:e_united_kingdom.holder }
		}
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = wm_found_change
			add_claim_on_loss = no
		}
		change_liege = {
			liege = title:e_united_kingdom.holder
			change = scope:wm_found_change
		}
		resolve_title_and_vassal_change = scope:wm_found_change
	}
	# Vanilla's on_dynasty_created renames new houses after a barony unless this is set
	set_variable = { name = block_dynamic_dynasty_naming value = yes }
	create_dynasty = {
		name = wm_party_$SLOT$_house
		coat_of_arms = wm_coa_party_$SLOT$
		spread_to_descendants = no
		save_scope_as = wm_found_dynasty
	}
	house ?= {
		set_house_name = wm_party_$SLOT$_house
		set_variable = { name = wm_house_party value = scope:wm_found_party }
	}
	get_title = scope:wm_found_party
	scope:wm_found_party = {
		add_title_law = wm_party_succession_law
	}
	set_variable = { name = wm_party value = scope:wm_found_party }
	wm_party_fresh_start_effect = yes
}
