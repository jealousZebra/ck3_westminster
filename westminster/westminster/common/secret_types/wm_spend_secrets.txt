# Campaign spending (common/scripted_effects/wm_spend_effects.txt): a smear campaign that backfired. The press
# knows who paid for it; a rival who learns the details can expose it.
wm_secret_smear_campaign = {
	category = civil

	is_valid = {
		scope:secret_owner = { is_alive = yes }
	}

	is_shunned = {
		always = yes
	}

	is_criminal = {
		always = no
	}

	on_expose = {
		scope:secret_owner = {
			add_prestige = medium_prestige_loss
			if = {
				limit = { exists = var:wm_party }
				set_global_variable = { name = wm_campaign_amount value = -5 }
				wm_campaign_party_poll_effect = { PARTY = var:wm_party }
			}
		}
		wm_trust_scandal_effect = { AMOUNT = wm_trust_scandal_exposed }
	}
}
