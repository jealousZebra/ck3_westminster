# The Prime Minister devolves a power to Scotland, Wales or Northern Ireland.
wm_devolution_devolve_decision = {
	picture = {
		reference = "gfx/interface/illustrations/loading_screens/wm_senedd.dds"
	}
	decision_group_type = major
	sort_order = 50
	ai_check_interval = 24

	desc = wm_devolution_devolve_decision_desc
	selection_tooltip = wm_devolution_devolve_decision_tooltip

	is_shown = {
		has_title = title:e_united_kingdom
	}

	cooldown = { years = 1 }

	cost = {
		prestige = wm_dev_powers_cost
	}

	effect = {
		custom_tooltip = wm_devolution_devolve_decision_effect_tt
		hidden_effect = {
			trigger_event = wm_devolution.20
		}
	}

	ai_potential = {
		prestige >= 500
	}

	# Buy off rising independence sentiment
	ai_will_do = {
		base = 0
		modifier = {
			add = 30
			OR = {
				wm_ui_dev_support_scotland >= 50
				wm_ui_dev_support_wales >= 50
				wm_ui_dev_support_northern_ireland >= 50
			}
		}
	}
}
