# A wing moves against its leader (wm_wing_challenge_effect). scope:wm_wing_challenger was saved by the caller;
# the global variables carry it too.
wm_wing.10 = {
	type = character_event
	title = wm_wing.10.t
	desc = wm_wing.10.desc
	theme = party
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = global_var:wm_wing_challenger
		animation = anger
	}
	trigger = {
		exists = global_var:wm_wing_challenger
	}
	immediate = {
		global_var:wm_wing_challenger = { save_scope_as = wm_wing_challenger }
	}

	option = {
		name = wm_wing.10.a
		custom_tooltip = wm_wing.10.a.tt
		wm_wing_concede_effect = yes
	}
	option = {
		name = wm_wing.10.b
		custom_tooltip = wm_wing.10.b.tt
		wm_wing_face_down_effect = yes
	}
}
